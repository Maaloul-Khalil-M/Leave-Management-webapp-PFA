package com.stagepfa.demo.migrations;

import com.mongodb.client.MongoCollection;
import com.mongodb.client.MongoDatabase;
import io.flamingock.api.annotations.Apply;
import io.flamingock.api.annotations.Change;
import io.flamingock.api.annotations.Rollback;
import io.flamingock.api.annotations.TargetSystem;
import org.bson.Document;

import java.time.Instant;
import java.time.LocalDate;
import java.time.Year;
import java.util.List;

@TargetSystem(id = "leave-management-mongo")
@Change(id = "seed-leave-foundation", author = "stagepfa")
public class _0003__SeedLeaveFoundation {

    // Reuse IDs from _0002__SeedEmployeesAndUsers
    public static final String EMP_AHMED_ID = "665f00000000000000000030";
    public static final String EMP_SALMA_ID = "665f00000000000000000031";

    public static final String ORG_SETTINGS_ID     = "665f00000000000000000050";
    public static final String LEAVE_TYPE_ANNUAL_ID = "665f00000000000000000060";
    public static final String LEAVE_TYPE_SICK_ID   = "665f00000000000000000061";
    public static final String LEAVE_TYPE_UNPAID_ID = "665f00000000000000000062";

    public static final String POLICY_TN_ANNUAL_ID = "665f00000000000000000070";
    public static final String POLICY_TN_SICK_ID   = "665f00000000000000000071";

    public static final String LEDGER_AHMED_ANNUAL_ID = "665f00000000000000000080";
    public static final String LEDGER_SALMA_ANNUAL_ID = "665f00000000000000000081";

    private static final int CURRENT_YEAR = Year.now().getValue();

    @Apply
    public void apply(MongoDatabase db) {
        seedOrganizationSettings(db);
        seedLeaveTypes(db);
        seedLeavePolicies(db);
        seedLeaveLedgers(db);
        // leave_requests intentionally empty — created via API
    }

    @Rollback
    public void rollback(MongoDatabase db) {
        db.getCollection("leave_ledgers").deleteMany(new Document());
        db.getCollection("leave_policies").deleteMany(new Document());
        db.getCollection("leave_types").deleteMany(new Document());
        db.getCollection("organization_settings").deleteMany(new Document());
        // leave_requests left untouched on rollback of this change
    }

    // -------------------------------------------------------------------------
    // organization_settings (singleton)
    // -------------------------------------------------------------------------

    private void seedOrganizationSettings(MongoDatabase db) {
        MongoCollection<Document> col = db.getCollection("organization_settings");
        Instant now = Instant.now();

        // weekendDays: ISO-8601 — 1=Mon … 7=Sun → Saturday=6, Sunday=7
        col.insertOne(new Document("_id", ORG_SETTINGS_ID)
                .append("companyName", "Acme Tunisie SARL")
                .append("country", "TN")
                .append("timezone", "Africa/Tunis")
                .append("weekendDays", List.of(6, 7))
                .append("updatedAt", now)
                .append("updatedBy", "admin@acme.tn"));
    }

    // -------------------------------------------------------------------------
    // leave_types
    // -------------------------------------------------------------------------

    private void seedLeaveTypes(MongoDatabase db) {
        MongoCollection<Document> col = db.getCollection("leave_types");

        col.insertOne(new Document("_id", LEAVE_TYPE_ANNUAL_ID)
                .append("code", "PAID_ANNUAL")
                .append("label", "Congé Payé")
                .append("legalCategory", "ANNUAL_LEAVE")
                .append("requiresProof", false)
                .append("deductsFromBalance", true)
                .append("active", true));

        col.insertOne(new Document("_id", LEAVE_TYPE_SICK_ID)
                .append("code", "SICK")
                .append("label", "Congé Maladie")
                .append("legalCategory", "SICKNESS")
                .append("requiresProof", true)
                .append("deductsFromBalance", false)
                .append("active", true));

        col.insertOne(new Document("_id", LEAVE_TYPE_UNPAID_ID)
                .append("code", "UNPAID")
                .append("label", "Congé Sans Solde")
                .append("legalCategory", "UNPAID_ABSENCE")
                .append("requiresProof", false)
                .append("deductsFromBalance", false)
                .append("active", true));
    }

    // -------------------------------------------------------------------------
    // leave_policies (TN only for now)
    // -------------------------------------------------------------------------

    private void seedLeavePolicies(MongoDatabase db) {
        MongoCollection<Document> col = db.getCollection("leave_policies");

        // Tunisia paid annual: 1 working day / month, min block 6 days (simplified)
        col.insertOne(new Document("_id", POLICY_TN_ANNUAL_ID)
                .append("country", "TN")
                .append("leaveTypeCode", "PAID_ANNUAL")
                .append("effectiveFrom", LocalDate.of(2024, 1, 1))
                .append("effectiveTo", null)
                .append("accrualUnit", "WORKING_DAY")
                .append("accrualRate", 1.0)
                .append("maxBalance", null)
                .append("minBlockDays", 6.0)
                .append("noticeDays", 7)
                .append("active", true));

        // Sick — no balance deduction
        col.insertOne(new Document("_id", POLICY_TN_SICK_ID)
                .append("country", "TN")
                .append("leaveTypeCode", "SICK")
                .append("effectiveFrom", LocalDate.of(2024, 1, 1))
                .append("effectiveTo", null)
                .append("accrualUnit", "CALENDAR_DAY")
                .append("accrualRate", 0.0)
                .append("maxBalance", null)
                .append("minBlockDays", null)
                .append("noticeDays", 0)
                .append("active", true));
    }

    // -------------------------------------------------------------------------
    // leave_ledgers (current year, PAID_ANNUAL for Ahmed + Salma)
    // -------------------------------------------------------------------------

    private void seedLeaveLedgers(MongoDatabase db) {
        MongoCollection<Document> col = db.getCollection("leave_ledgers");
        Instant accrualDate = Instant.parse(CURRENT_YEAR + "-01-31T00:00:00Z");

        // Ahmed: 8 months accrued example → 8.0 available
        col.insertOne(buildLedger(
                LEDGER_AHMED_ANNUAL_ID,
                EMP_AHMED_ID,
                8.0,
                accrualDate,
                "Initial seed accrual (Jan–Aug)"));

        // Salma: 12 months → 12.0 available
        col.insertOne(buildLedger(
                LEDGER_SALMA_ANNUAL_ID,
                EMP_SALMA_ID,
                12.0,
                accrualDate,
                "Initial seed accrual (full year to date)"));
    }

    private Document buildLedger(String id,
                                 String employeeId,
                                 double accrued,
                                 Instant movementDate,
                                 String note) {
        Document movement = new Document()
                .append("date", movementDate)
                .append("type", "MONTHLY_ACCRUAL")
                .append("amount", accrued)
                .append("note", note)
                .append("leaveRequestId", null)
                .append("actorUserId", null);

        return new Document("_id", id)
                .append("employeeId", employeeId)
                .append("leaveTypeCode", "PAID_ANNUAL")
                .append("year", CURRENT_YEAR)
                .append("policyId", POLICY_TN_ANNUAL_ID)
                .append("accruedToDate", accrued)
                .append("consumedBalance", 0.0)
                .append("carriedOverFromPreviousYear", 0.0)
                .append("availableBalance", accrued)
                .append("movements", List.of(movement));
    }
}
